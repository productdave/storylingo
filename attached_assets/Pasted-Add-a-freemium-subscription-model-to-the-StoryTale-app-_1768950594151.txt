7. Subscription Status Check
Before playing any story, check:
javascriptasync function canPlayStory(storyId) {
  const userData = await getUserData();
  
  // Premium users can play anything
  if (userData.subscriptionStatus !== 'free') {
    return { allowed: true };
  }
  
  // Free users: check monthly limit
  if (userData.storiesThisMonth >= userData.freeStoriesLimit) {
    return { 
      allowed: false, 
      reason: 'limit_reached',
      showPaywall: true 
    };
  }
  
  // Check if story requires premium
  if (isPremiumStory(storyId)) {
    return { 
      allowed: false, 
      reason: 'premium_only',
      showPaywall: true 
    };
  }
  
  return { allowed: true };
}
8. Story Counter Increment
After user starts listening to a story:
javascriptasync function incrementStoryCount() {
  const userData = await getUserData();
  
  // Only count for free users
  if (userData.subscriptionStatus === 'free') {
    userData.storiesThisMonth += 1;
    await saveUserData(userData);
  }
}
9. Monthly Reset Logic
Implement automatic counter reset:
javascriptasync function checkAndResetMonthlyCounter() {
  const userData = await getUserData();
  const today = new Date();
  const resetDate = new Date(userData.monthResetDate);
  
  if (today >= resetDate) {
    // Reset counter
    userData.storiesThisMonth = 0;
    
    // Set next reset date (first of next month)
    const nextMonth = new Date(today.getFullYear(), today.getMonth() + 1, 1);
    userData.monthResetDate = nextMonth.toISOString();
    
    await saveUserData(userData);
  }
}
